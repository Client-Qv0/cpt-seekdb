class ContainerProcess:
    def __init__(self,proc,name):
        self.proc=proc;self.name=name
        self.stdin=proc.stdin;self.stdout=proc.stdout
    def poll(self):return self.proc.poll()
    def wait(self,timeout=None):return self.proc.wait(timeout=timeout)
    def kill(self):
        # Kill the actual database container, not merely the attached Docker CLI.
        deadline=time.monotonic()+8
        while time.monotonic()<deadline and self.proc.poll() is None:
            found=subprocess.run(['docker','inspect','--format','{{.State.Status}}',self.name],capture_output=True,text=True)
            if found.returncode==0:
                subprocess.run(['docker','kill','--signal','KILL',self.name],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
                subprocess.run(['docker','rm','--force',self.name],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
                return
            time.sleep(.02)
        if self.proc.poll() is None:
            self.proc.kill()
            subprocess.run(['docker','rm','--force',self.name],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
