cmake_minimum_required(VERSION 3.16)
project(MiniVectorDB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mvdb src/sql.cpp src/storage.cpp src/engine.cpp src/spill.cpp src/protocol.cpp)
target_include_directories(mvdb PUBLIC src)
target_compile_options(mvdb PRIVATE -Wall -Wextra -Wpedantic)
add_executable(minivectordb src/main.cpp)
target_link_libraries(minivectordb PRIVATE mvdb)
enable_testing()
foreach(unit parser storage engine_extra spill compact_rows protocol)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${unit}.cpp")
    add_executable(test_${unit} tests/test_${unit}.cpp)
    target_link_libraries(test_${unit} PRIVATE mvdb)
    add_test(NAME ${unit} COMMAND test_${unit})
  endif()
endforeach()
